   ```ts
   // Client-side sort allowlist — URL/UI domain.
   // Server-side counterpart is SORT_COLUMNS in lib/db/queries.ts (SQL ORDER BY allowlist).
   // Keep arrays in sync; intentionally duplicated to keep better-sqlite3 out of the client bundle.
   export const CLIENT_SORT_KEYS = ['newest', 'price_asc', 'price_desc', 'name_asc'] as const
   export type ClientSortKey = (typeof CLIENT_SORT_KEYS)[number]

   export function isClientSortKey(v: string | null | undefined): v is ClientSortKey {
   ```

2. **`lib/image.ts`** (new — same-origin guard for `next/image`):
   ```ts
   const PLACEHOLDER = '/seed-images/_placeholder.svg' // ships in public/ (see Phase A note)
   // Defense-in-depth: product.image_url is a free-form string in the DB. Reject anything
   // that isn't a same-origin path to prevent (a) future remotePatterns leak from sourcing
   // attacker-controlled bytes through the image optimizer, (b) protocol-relative URLs.
   export function safeProductImage(src: string | undefined | null): string {

     return src
   }
   ```

4. **`lib/hooks/useProducts.ts`** (`'use client'`):
   ```ts
   'use client'

   import { useQuery } from '@tanstack/react-query'
   import { HTTPError } from 'ky'
   import { api } from '@/lib/api-client'
   import type { Product, ProductListQuery, ApiError } from '@/lib/types'

   function buildSearchParams(q: ProductListQuery): URLSearchParams {
     const sp = new URLSearchParams()
     if (q.category) sp.set('category', q.category)
     if (q.sort) sp.set('sort', q.sort)
     if (q.q) sp.set('q', q.q)
     if (q.limit !== undefined) sp.set('limit', String(q.limit))
     if (q.offset !== undefined) sp.set('offset', String(q.offset))
     return sp
   }

   // Strip undefined fields for stable key hashing across {a:undefined} vs {}.
   function normalizeKey(q: ProductListQuery): Record<string, string | number> {
     const out: Record<string, string | number> = {}
     if (q.category) out.category = q.category
     if (q.sort) out.sort = q.sort
     if (q.q) out.q = q.q
     if (q.limit !== undefined) out.limit = q.limit
     if (q.offset !== undefined) out.offset = q.offset
     return out
   }

   export function useProducts(params: ProductListQuery) {
     return useQuery({
       queryKey: ['products', normalizeKey(params)] as const,
       queryFn: async () => {
         try {
           return await api.get('products', { searchParams: buildSearchParams(params) }).json<Product[]>()
         } catch (e) {
           if (e instanceof HTTPError) {
             const body = await e.response.json<ApiError>().catch(() => ({ error: e.message }))
             // Surface raw server message in dev only — production renders a generic string in the UI.
             const message = process.env.NODE_ENV === 'production'
               ? (e.response.status >= 500 ? 'Could not load products' : 'Bad request')
               : (body.error || `HTTP ${e.response.status}`)
             const err = new Error(message)
             ;(err as Error & { status?: number }).status = e.response.status
             throw err
           }
           throw e
         }
       },
     })
   }
   ```
   - Structured query key (`['products', { ... }]`) — TanStack v5 hashes objects deterministically and supports prefix-matching for partial invalidation. Future Wk7/8 mutations can do `queryClient.invalidateQueries({ queryKey: ['products'] })` to clear all catalog lists, or scoped predicates for specific filters.
   - Error message sanitized in production (no DB hint leaks); raw message preserved in dev for debugging.
   - Status attached to the error for future consumers (e.g. Wk7 mutations may want to distinguish 401/409 — typed `ApiClientError` contract is **deferred** to Wk7 prep; see §4).

### Phase E — `FilterBar` component

**Commit subject:** `feat(ui): URL-driven FilterBar w/ debounced search`

**File: `components/FilterBar.tsx`** (`'use client'`):

(`lib/sort.ts` already shipped in Phase B — `FilterBar` consumes it.)

```tsx
'use client'

import { useEffect, useRef, useState } from 'react'
import { usePathname, useRouter, useSearchParams } from 'next/navigation'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select'
import { Input } from '@/components/ui/input'
import { useCategories } from '@/lib/hooks/useCategories'
import { CLIENT_SORT_KEYS, isClientSortKey, type ClientSortKey } from '@/lib/sort'

const SEARCH_DEBOUNCE_MS = 250
const SEARCH_MAX_LEN = 100
const ALL_VALUE = '__all__'

const SORT_OPTIONS: { value: ClientSortKey; label: string }[] = [
  { value: 'newest',     label: 'Newest' },
  { value: 'price_asc',  label: 'Price ↑' },
  { value: 'price_desc', label: 'Price ↓' },
  { value: 'name_asc',   label: 'Name A–Z' },
]

export function FilterBar() {
  const router = useRouter()
  const pathname = usePathname()
  const searchParams = useSearchParams()

  const category = searchParams.get('category') ?? ''
  const rawSort = searchParams.get('sort')
  const sort: ClientSortKey = isClientSortKey(rawSort) ? rawSort : 'newest'
  const urlQ = searchParams.get('q') ?? ''

  const [qInput, setQInput] = useState(urlQ)
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)

  // Sync local input when URL `q` changes externally (back/forward, clear-filters).
  useEffect(() => {
    setQInput(urlQ)
  }, [urlQ])

  // pathname is Next-derived (usePathname), never user-controlled — safe to interpolate.
  function setParam(name: string, value: string) {
    const next = new URLSearchParams(searchParams.toString())
    if (value === '') next.delete(name)
    else next.set(name, value)
    const qs = next.toString()
    router.replace(qs ? `${pathname}?${qs}` : pathname, { scroll: false })
  }

  function onSearchChange(value: string) {
    const capped = value.slice(0, SEARCH_MAX_LEN)
    setQInput(capped)
    if (debounceRef.current) clearTimeout(debounceRef.current)
    debounceRef.current = setTimeout(() => setParam('q', capped.trim()), SEARCH_DEBOUNCE_MS)
  }

  useEffect(() => {
    return () => {
      if (debounceRef.current) clearTimeout(debounceRef.current)
    }
  }, [])

  const { data: categories } = useCategories()
  const categoryOptions = categories ?? []

  return (
    <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
      <Select
        value={category === '' ? ALL_VALUE : category}
        onValueChange={(v) => setParam('category', v === ALL_VALUE ? '' : v)}
      >
        <SelectTrigger className="w-full sm:w-48"><SelectValue placeholder="All categories" /></SelectTrigger>
        <SelectContent>
          <SelectItem value={ALL_VALUE}>All categories</SelectItem>
          {categoryOptions.map((c) => (
            <SelectItem key={c} value={c}>{c}</SelectItem>
          ))}
        </SelectContent>
      </Select>

      <Select value={sort} onValueChange={(v) => setParam('sort', v === 'newest' ? '' : v)}>
        <SelectTrigger className="w-full sm:w-40"><SelectValue /></SelectTrigger>
        <SelectContent>
          {SORT_OPTIONS.map((o) => (
            <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>
          ))}
        </SelectContent>
      </Select>

      <Input
        type="search"
        placeholder="Search products…"
        value={qInput}
        onChange={(e) => onSearchChange(e.target.value)}
        maxLength={SEARCH_MAX_LEN}
        className="w-full sm:w-64"
      />
    </div>
  )
}
```

**Notes / decisions baked in:**
- **No `lib/db/queries` import here.** `lib/sort.ts` (Phase E step 1) is the client-safe sort source-of-truth; importing the DB module would pull `better-sqlite3` (native) into the client bundle and explode `bun run build`.
- "All" pushes `''` → `setParam` deletes the key. `'newest'` is the default sort → `setParam('sort', '')` deletes the key (URL stays clean for the default).
- Debounce cleanup on unmount (prevents a stale `setTimeout` firing after the user navigates away and triggering an unmount-time `router.replace`).
- `value={qInput}` controlled input — local state for in-flight typing, URL syncs on settle.
- `useEffect([urlQ])` syncs local input when URL is mutated externally (e.g. back-button, clear-filters).
- Pure `router.replace` (not `push`) so each keystroke / dropdown change does not pollute history. Back/forward step over the *committed* states only — exactly what the spec calls for.

```tsx
'use client'

import { usePathname, useRouter, useSearchParams } from 'next/navigation'
import { useProducts } from '@/lib/hooks/useProducts'
import { useCategories } from '@/lib/hooks/useCategories'
import { ProductCard } from '@/components/ProductCard'
import { ProductGridSkeleton } from '@/components/ProductGridSkeleton'
import { Button } from '@/components/ui/button'
import { isClientSortKey, type ClientSortKey } from '@/lib/sort'
import type { Product } from '@/lib/types'

const PAGE_LIMIT = 24

function readSort(v: string | null): ClientSortKey | undefined {
  return isClientSortKey(v) ? v : undefined
}

export function ProductGrid() {
  const sp = useSearchParams()
  const router = useRouter()
  const pathname = usePathname()

  const { data: categories } = useCategories()
  const rawCategory = sp.get('category') || undefined
  // Drop unknown category values — defends against crafted URLs and stale links.
  // While categories are loading, treat as undefined (no filter) rather than blocking the grid.
  const category = rawCategory && categories?.includes(rawCategory) ? rawCategory : undefined

  const sort = readSort(sp.get('sort'))
  const q = sp.get('q') || undefined

  // Spec §"Empty state": "when q || category is set". Clear filters strips ALL params
  // (including sort) — simpler to reason about than per-axis preservation.
  const filtersActive = Boolean(category || q)
  const { data, isPending, isError, error, refetch } = useProducts({ category, sort, q, limit: PAGE_LIMIT })

  if (isPending) return <ProductGridSkeleton />
  if (isError) return <ErrorState message={error.message} onRetry={() => refetch()} />
  if (!data?.length) {
    return filtersActive
      ? <FilteredEmpty onClear={() => router.replace(pathname)} />
      : <EmptyState />
  }
  return <Grid items={data} />
}

function Grid({ items }: { items: Product[] }) {
  return (
    <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4">
      {items.map((p) => <ProductCard key={p.id} product={p} />)}
    </div>
  )
}

function ErrorState({ message, onRetry }: { message: string; onRetry: () => void }) {
  return (
    <div className="flex flex-col items-center gap-3 py-16 text-center">
      <p className="text-sm text-muted-foreground">{message}</p>
      <Button onClick={onRetry}>Retry</Button>
    </div>
  )
}

function FilteredEmpty({ onClear }: { onClear: () => void }) {
  return (
    <div className="flex flex-col items-center gap-3 py-16 text-center">
      <p className="text-sm text-muted-foreground">No products match your filters.</p>
      <Button variant="outline" onClick={onClear}>Clear filters</Button>
    </div>
  )
}

function EmptyState() {
  return (
    <div className="py-16 text-center text-sm text-muted-foreground">
      No products yet.
    </div>
  )
}
```

## 4. Risks / open notes (deferred)

- **Suspense boundary placement:** boundary wraps `<ProductGrid />` only — `<FilterBar />` is outside it, so input state survives across re-fetches. Future pages that use `useSearchParams` must add their own `<Suspense>` (build will fail otherwise).
- **`SORT_KEYS` server↔client duplication:** server-side `SORT_COLUMNS` (in `lib/db/queries.ts`) is the SQL allowlist; client-side `CLIENT_SORT_KEYS` (in `lib/sort.ts`) is the URL/UI allowlist. Tiny string arrays, intentionally duplicated to keep the DB module out of the client bundle. Drift detection: server falls through to default sort on unknown values (silent), client `isClientSortKey` rejects unknown values (silent). For louder failure mode, future Wk7/8 server route can return 400 on unknown `sort` instead of falling through.
- **Spec deviation: Header link:** Wk3 spec §6 specifies plain `<a>`; Phase C uses `next/link` for SPA navigation parity with `ProductCard`. Recorded for transparency.
- **Categories cache invalidation:** demo's catalog is static, no admin UI. If Wk7/8 adds product CRUD, `queryClient.invalidateQueries({ queryKey: ['categories'] })` on mutation success.
- **Pagination:** out-of-scope. Limit fixed at 24, server cap 50.
- **Detail page (`/products/[slug]`):** Wk5. Cards link there now → 404 expected.
- **Typed error contract for `lib/hooks/*` (deferred to Wk7 prep):** `useProducts` currently throws `Error` with `status` patched on. Wk7 cart mutations need richer typing (`ApiClientError` class with `status`/`body`) and likely a `ky` `beforeError` hook in `lib/api-client.ts` so all hooks share the contract. Deferred to keep Wk3/4 scope tight.
- **Server-side caching on `/api/products/categories`:** currently `dynamic = 'force-dynamic'` + `Cache-Control: no-store` (consistent with `/api/products`). Categories rarely change; once Wk7/8 admin mutations land, switch to `revalidate = 300` with `revalidatePath` on category-changing mutations. Out of scope here.
- **Unauthenticated rate-limiting:** `/api/products` and `/api/products/categories` are public, uncapped. Search endpoint is the cheapest amplifier (full-table LIKE scan). Demo accepts the risk; a real deployment should add a token bucket / edge limit before exposure. The Wk3/4 plan adds a 100-char cap on `q` at the FilterBar write-side as defense-in-depth (server cap deferred — `app/api/products/route.ts` is do-not-touch in this plan).
- **Linting layer rule extension (deferred):** add oxlint `no-restricted-imports` to forbid `@/lib/db/**` from `lib/types.ts`, `lib/hooks/**`, `lib/sort.ts`, `components/**`. Currently policed by reviewer + grep verify steps. Codify in oxlint config in a later cleanup commit.
- **Server `ListProductsFilter` vs client `ProductListQuery` shape asymmetry:** server type makes `limit`/`offset` required (per `lib/db/queries.ts:25-31`), client type makes them optional. Intentional — the route handler always supplies defaults before calling `listProducts`, so the client doesn't need to. Recorded so future readers don't try to "align" the two.
- **Phase A/D shadcn-add commits mix generated + authored files:** acceptable grain; could be split into `chore(ui): add shadcn …` + `feat: …` commits if bisectability becomes a concern. Not worth the phase-count expansion now.
